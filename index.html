/* ========================================================= MASTER QUANT CORE — REFACTORED ANALYSIS ENGINE --------------------------------------------------------- CORE: 5 Main Models SUPPORT: 100 Formula Engine Radar Layer Backend Validation Win Candidate DISPLAY: Confidence = fixed 96.9% Accuracy = 50.1% - 50.9% IMPORTANT: Display metrics NEVER modify Core Formula outputs. ========================================================= */ const QUANT_CONFIG = Object.freeze({ DISPLAY_CONFIDENCE: 96.9, ACCURACY_MIN: 50.1, ACCURACY_MAX: 50.9, WEIGHTS: Object.freeze({ CORE: 0.55, FORMULA100: 0.20, RADAR: 0.10, BACKEND: 0.10, WIN: 0.05 }) }); function normalizeDigit(value) { const n = Number(value); return ((n % 10) + 10) % 10; } function uniqueDigits(arr) { return [...new Set( arr .map(Number) .filter(Number.isInteger) .map(normalizeDigit) )]; } function clamp(value, min, max) { return Math.min(max, Math.max(min, value)); } function seededAccuracy() { /* Display-only metric. Never feeds back into scoring. */ const range = QUANT_CONFIG.ACCURACY_MAX - QUANT_CONFIG.ACCURACY_MIN; return Number( ( QUANT_CONFIG.ACCURACY_MIN + Math.random() * range ).toFixed(1) ); } /* ========================================================= 5 MAIN FORMULAS ========================================================= */ function buildFiveMainModels(H, T, U, R1, R2) { const sumTop = H + T + U; const sumBottom = R1 + R2; const raw = [ (T + R1) % 10, sumTop % 10, (H + R1) % 10, (U + R2) % 10, (10 - R2) % 10 ]; const definitions = [ { id: 1, name: "AI โมเดล 01", code: "M1", description: "T + R1 → หลักหน่วย" }, { id: 2, name: "AI โมเดล 02", code: "M2", description: "H + T + U → หลักหน่วย" }, { id: 3, name: "AI โมเดล 03", code: "M3", description: "H + R1 → หลักหน่วย" }, { id: 4, name: "AI โมเดล 04", code: "M4", description: "U + R2 → หลักหน่วย" }, { id: 5, name: "AI โมเดล 05", code: "M5", description: "10 - R2 → หลักหน่วย" } ]; return definitions.map((model, index) => { const primary = normalizeDigit(raw[index]); const secondary = normalizeDigit(primary + 1); return { ...model, rawValue: raw[index], set: uniqueDigits([ primary, secondary ]), primary, secondary, weight: 1 }; }); } /* ========================================================= 100-FORMULA SUPPORT ENGINE ========================================================= */ function buildSupportFormulaEngine( H, T, U, R1, R2 ) { const sumTop = H + T + U; const sumBottom = R1 + R2; const total = sumTop + sumBottom; const formulas = [ H + T, H + U, T + U, H + R1, H + R2, T + R1, T + R2, U + R1, U + R2, R1 + R2, sumTop + R1, sumTop + R2, sumTop + H, sumTop + T, sumTop + U, 10 - H, 10 - T, 10 - U, 10 - R1, 10 - R2, H + T + R1, H + U + R2, T + U + R1, T + U + R2, H + R1 + R2, T + R1 + R2, sumTop - R1, sumTop - R2, sumTop + sumBottom, sumTop - sumBottom, H + T - R2, T + U - R1, H + U - R1, 20 - (T + R1), 20 - (U + R2), H + T + U - R1, H + T + U - R2, 10 + T - R2, 10 + U - R1, sumTop + 10 - R2, H + R2 - T, T + R2 - H, U + R1 - H, sumBottom + H - T, sumBottom + H - U, H + sumBottom - T, H + sumBottom - U, T + sumBottom - H, T + sumBottom - U, U + sumBottom - H, U + sumBottom - T, 30 - (H + T + R1), 30 - (H + U + R2), 30 - (T + U + R1), 30 - (H + R1 + R2), H + T + R1 - R2, H + U + R2 - R1, T + U + R1 - H, sumTop + R1 - R2, sumTop + R2 - R1, H + T + sumBottom, H + U + sumBottom, T + U + sumBottom, H + R1 + sumTop, H + R2 + sumTop, T + R1 + sumTop, T + R2 + sumTop, U + R1 + sumTop, U + R2 + sumTop, R1 + R2 + sumTop, 10 + H + T, 10 + H + U, 10 + T + U, 10 + R1 + R2, 10 + H + R1, 10 + H + R2, 10 + T + R1, 10 + T + R2, 10 + U + R1, 10 + U + R2, H - T + R2, H - U + R1, T - U + R2, R1 - R2 + H, R1 - R2 + T, R1 - R2 + U, 20 - H - T, 20 - H - U, 20 - T - U, 20 - R1 - R2, 20 - H - R1, 20 - H - R2, 20 - T - R1, 20 - T - R2, 20 - U - R1, 20 - U - R2, total, total * 2, total + H, total + R1 ]; const counts = {}; for (let d = 0; d <= 9; d++) { counts[d] = 0; } formulas.forEach(value => { const digit = normalizeDigit(value); counts[digit]++; }); return { formulas, counts, formulaCount: formulas.length }; } /* ========================================================= RADAR SUPPORT ========================================================= */ function buildRadarScores(H, T, U, R1, R2) { const x = normalizeDigit(R1 + R2); const y = normalizeDigit(H + T + U); const z = normalizeDigit(H + T + U + R1 + R2); const ranges = [ [ normalizeDigit(x - 1), x, normalizeDigit(x + 1) ], [ normalizeDigit(y - 1), y, normalizeDigit(y + 1) ], [ normalizeDigit(z - 1), z, normalizeDigit(z + 1) ] ]; const scores = {}; for (let d = 0; d <= 9; d++) { let hits = 0; ranges.forEach(range => { if (range.includes(d)) hits++; }); scores[d] = hits / ranges.length; } return scores; } /* ========================================================= BACKEND VALIDATION ========================================================= */ function buildBackendScores(H, T, U, R1, R2) { const sumTop = H + T + U; const total = sumTop + R1 + R2; const candidates = uniqueDigits([ 2 * H + R1, T + R2, sumTop, Math.abs(T - U), total ]); const scores = {}; for (let d = 0; d <= 9; d++) { scores[d] = candidates.includes(d) ? 1 : 0; } return scores; } /* ========================================================= WIN CANDIDATE ========================================================= */ function buildWinScores(H, T, U, R1, R2) { const candidate = normalizeDigit(H + T + U + R1 + R2); const scores = {}; for (let d = 0; d <= 9; d++) { scores[d] = d === candidate || d === normalizeDigit(candidate + 5) ? 1 : 0; } return scores; } /* ========================================================= DIGIT SCORE ENGINE ========================================================= */ function buildDigitMatrix( models, support, radar, backend, win ) { const modelHits = {}; for (let d = 0; d <= 9; d++) { modelHits[d] = 0; } models.forEach(model => { model.set.forEach(digit => { modelHits[digit] = (modelHits[digit] || 0) + 1; }); }); const maxSupport = Math.max( ...Object.values(support.counts), 1 ); const matrix = []; for (let d = 0; d <= 9; d++) { const coreScore = modelHits[d] / models.length; const supportScore = support.counts[d] / maxSupport; const radarScore = radar[d] || 0; const backendScore = backend[d] || 0; const winScore = win[d] || 0; /* CORE is intentionally dominant. */ const finalScore = coreScore * QUANT_CONFIG.WEIGHTS.CORE + supportScore * QUANT_CONFIG.WEIGHTS.FORMULA100 + radarScore * QUANT_CONFIG.WEIGHTS.RADAR + backendScore * QUANT_CONFIG.WEIGHTS.BACKEND + winScore * QUANT_CONFIG.WEIGHTS.WIN; matrix.push({ digit: d, coreHits: modelHits[d], coreScore, supportScore, radarScore, backendScore, winScore, totalScore: finalScore }); } return matrix.sort( (a, b) => b.totalScore - a.totalScore || b.coreHits - a.coreHits || b.supportScore - a.supportScore || b.digit - a.digit ); } /* ========================================================= SCORE → DISPLAY PERCENT ========================================================= */ function scoreToPercent(matrix, target) { const scores = matrix.map(item => item.totalScore); const min = Math.min(...scores); const max = Math.max(...scores); const range = max - min || 1; const normalized = (target.totalScore - min) / range; return Number( ( 82 + normalized * 15.5 ).toFixed(1) ); } /* ========================================================= PAIR GENERATOR ========================================================= */ function buildPairs(sourceDigit, rankedMatrix) { const pairs = []; for (const item of rankedMatrix) { const target = item.digit; if (target === sourceDigit) { continue; } const pair = `${sourceDigit}${target}`; if (!pairs.some(x => x.pair === pair)) { pairs.push({ pair, target, score: item.totalScore }); } if (pairs.length >= 5) { break; } } return pairs; } /* ========================================================= WIN SET 6 ========================================================= */ function buildWinSet(matrix, models) { const result = []; /* First priority: digits appearing in the 5 Main Models. */ const coreDigits = new Set(); models.forEach(model => { model.set.forEach(digit => { coreDigits.add(digit); }); }); matrix.forEach(item => { if ( coreDigits.has(item.digit) && result.length < 6 ) { result.push(item.digit); } }); /* Fill remaining slots from final ranking. */ matrix.forEach(item => { if ( result.length < 6 && !result.includes(item.digit) ) { result.push(item.digit); } }); return result .slice(0, 6) .sort((a, b) => a - b); } /* ========================================================= MAIN ANALYSIS ========================================================= */ function runMasterQuantAnalysis(n3, n2) { if (!/^\d{3}$/.test(n3)) { throw new Error("INVALID_3_DIGIT_INPUT"); } if (!/^\d{2}$/.test(n2)) { throw new Error("INVALID_2_DIGIT_INPUT"); } const [ H, T, U ] = n3.split("").map(Number); const [ R1, R2 ] = n2.split("").map(Number); /* ----------------------------------------------------- CORE ENGINE ----------------------------------------------------- */ const models = buildFiveMainModels( H, T, U, R1, R2 ); /* ----------------------------------------------------- SUPPORT ENGINES ----------------------------------------------------- */ const support = buildSupportFormulaEngine( H, T, U, R1, R2 ); const radar = buildRadarScores( H, T, U, R1, R2 ); const backend = buildBackendScores( H, T, U, R1, R2 ); const win = buildWinScores( H, T, U, R1, R2 ); /* ----------------------------------------------------- FINAL MATRIX ----------------------------------------------------- */ const matrix = buildDigitMatrix( models, support, radar, backend, win ); const primary = matrix[0]; const secondary = matrix.find( item => item.digit !== primary.digit ) || matrix[1]; const third = matrix.find( item => item.digit !== primary.digit && item.digit !== secondary.digit ) || matrix[2]; /* ----------------------------------------------------- DISPLAY METRICS ----------------------------------------------------- */ const primaryPercent = scoreToPercent( matrix, primary ); const secondaryPercent = scoreToPercent( matrix, secondary ); const thirdPercent = scoreToPercent( matrix, third ); /* IMPORTANT: This is DISPLAY ONLY. */ const confidence = QUANT_CONFIG.DISPLAY_CONFIDENCE; const accuracy = seededAccuracy(); /* ----------------------------------------------------- TOP PAIRS ----------------------------------------------------- */ const primaryPairs = buildPairs( primary.digit, matrix ); const secondaryPairs = buildPairs( secondary.digit, matrix ); /* ----------------------------------------------------- THREE-DIGIT CANDIDATES ----------------------------------------------------- */ const threeDigitCandidates = primaryPairs .slice(0, 5) .map(pair => `${pair.target}${primary.digit}${secondary.digit}` ); /* ----------------------------------------------------- WIN SET ----------------------------------------------------- */ const winSet = buildWinSet( matrix, models ); /* ----------------------------------------------------- MODEL DISPLAY SCORES ----------------------------------------------------- */ const modelResults = models.map(model => { const scoreSum = model.set.reduce( (sum, digit) => { const item = matrix.find( x => x.digit === digit ); return sum + (item ? item.totalScore : 0); }, 0 ); const avg = scoreSum / Math.max(model.set.length, 1); const modelDisplay = clamp( 82 + avg * 15.5, 82, 97.5 ); return { ...model, aiScore: Number( modelDisplay.toFixed(1) ) }; }); /* ----------------------------------------------------- EVEN / ODD ----------------------------------------------------- */ const evenCount = winSet.filter( d => d % 2 === 0 ).length; const oddCount = winSet.length - evenCount; const evenPercent = Number( ( evenCount / Math.max(winSet.length, 1) * 100 ).toFixed(1) ); const oddPercent = Number( ( 100 - evenPercent ).toFixed(1) ); return { input: { top3: n3, bottom2: n2 }, core: { models: modelResults }, support: { formula100: support, radar, backend, win }, ranking: matrix, primary, secondary, third, display: { confidence, accuracy, primaryPercent, secondaryPercent, thirdPercent }, pairs: { primary: primaryPairs, secondary: secondaryPairs }, threeDigitCandidates, winSet, balance: { evenPercent, oddPercent } }; }
