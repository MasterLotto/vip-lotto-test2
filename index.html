# ระบบวิเคราะห์และคำนวณสูตรหวย Version 2.8 (ปรับปรุงต่อเนื่องจาก Version 2.7) class LotteryEngineV28: def __init__(self): # โครงสร้างสูตรเดิมที่พัฒนาต่อจาก Version 2.7 พร้อมเพิ่มความเสถียรในการคำนวณ self.formulas = { "N1": lambda A, B, C, D, E: ((-2*A - 2*B + C + D - 2*E + 9) % 10, (-2*B + C + 3) % 10), "N6": lambda A, B, C, D, E: ((-A - 2*B + C - E + 1) % 10, (A - C - D - 2*E) % 10), "N1+V2": lambda A, B, C, D, E: ((A + 3*B + C + 3*D + 3*E + 6) % 10, (-A - B - C + D - 3*E + 1) % 10), "S1": lambda A, B, C, D, E: ((B + D) % 10, ((B + D) + 1) % 10), "S2": lambda A, B, C, D, E: ((A + B + C) % 10, ((A + B + C) + 1) % 10), "S3": lambda A, B, C, D, E: ((A + D) % 10, ((A + D) + 1) % 10), "S4": lambda A, B, C, D, E: ((C + E) % 10, ((C + E) + 1) % 10), "S5": lambda A, B, C, D, E: ((10 - E) % 10, ((10 - E) + 1) % 10), "S6": lambda A, B, C, D, E: ((A + E) % 10, ((A + E) + 1) % 10), "NEW": lambda A, B, C, D, E: ((B + E + 2) % 10, ((B + E + 2) + 2) % 10) } def calculate_next_draw(self, formula_name, top3, bottom2): if formula_name not in self.formulas: raise ValueError(f"ไม่พบสูตร {formula_name} ในระบบ") if len(top3) != 3 or len(bottom2) != 2: raise ValueError("รูปแบบข้อมูลรางวัลไม่ถูกต้อง (3 ตัวบนต้องมี 3 หลัก, 2 ตัวล่างต้องมี 2 หลัก)") A = int(top3[0]) B = int(top3[1]) C = int(top3[2]) D = int(bottom2[0]) E = int(bottom2[1]) formula_func = self.formulas[formula_name] digit1, digit2 = formula_func(A, B, C, D, E) return digit1, digit2 # ตัวอย่างการใช้งานสำหรับคำนวณงวดถัดไปจากฐาน Version 2.7 if __name__ == "__main__": engine = LotteryEngineV28() # สมมติผลรางวัลงวดล่าสุด 3 ตัวบน = 123, 2 ตัวล่าง = 45 d1, d2 = engine.calculate_next_draw("N1", "123", "45") print(f"ผลลัพธ์เลขเด่นจากสูตร N1: {d1} และ {d2}")
